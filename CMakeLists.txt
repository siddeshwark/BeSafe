cmake_minimum_required(VERSION 3.20)
project(AnisiddSafetyBackend LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(httplib GIT_REPOSITORY https://github.com/yhirose/cpp-httplib.git GIT_TAG v0.18.3)
FetchContent_Declare(json GIT_REPOSITORY https://github.com/nlohmann/json.git GIT_TAG v3.12.0)
FetchContent_MakeAvailable(httplib json)
find_package(CURL REQUIRED)
find_package(SQLite3 REQUIRED)
add_executable(anisidd-api main.cpp)
target_link_libraries(anisidd-api PRIVATE httplib::httplib nlohmann_json::nlohmann_json CURL::libcurl SQLite::SQLite3)
if(WIN32)
  target_link_libraries(anisidd-api PRIVATE ws2_32 crypt32)
endif()
